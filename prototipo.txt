let steps=[],stepIdx=0,tyTimer=null;
function openSteps(s){
  steps=s;stepIdx=0;showStep(0);
  document.getElementById('dlgOverlay').classList.add('open');
}
function showStep(i){
  const s=steps[i];
  document.getElementById('dlgPrt').textContent=s.p;
  document.getElementById('dlgName').textContent=s.n;
  document.getElementById('dlgBadge').innerHTML=s.badge||'';
  const el=document.getElementById('dlgTxt');
  el.textContent='';
  if(tyTimer)clearInterval(tyTimer);
  let ci=0;
  tyTimer=setInterval(()=>{if(ci<s.t.length){el.textContent+=s.t[ci++];}else clearInterval(tyTimer);},20);
  const ch=document.getElementById('dlgChoices');
  ch.innerHTML='';
  if(s.ch&&s.ch.length){
    document.getElementById('dlgNext').style.display='none';
    s.ch.forEach(c=>{const b=document.createElement('button');b.className='dch';b.textContent=c.l;b.onclick=()=>{clearInterval(tyTimer);c.cb();};ch.appendChild(b);});
  } else {document.getElementById('dlgNext').style.display='block';}
}
function nextDlg(){
  clearInterval(tyTimer);
  const s=steps[stepIdx];
  const el=document.getElementById('dlgTxt');
  if(el.textContent.length<s.t.length){el.textContent=s.t;return;}
  stepIdx++;
  if(stepIdx<steps.length)showStep(stepIdx);else closeDlg();
}
function closeDlg(){
  clearInterval(tyTimer);
  document.getElementById('dlgOverlay').classList.remove('open');
  steps=[];stepIdx=0;
}

  if(answered)return;answered=true;
  document.querySelectorAll('.abtn').forEach(b=>{
    if(b.textContent===correct)b.classList.add('correct');

    b.onclick=null;
  });
  const win=chosen===correct;
  const msg=document.createElement('div');
  msg.className='rmsg '+(win?'win':'lose');
  msg.innerHTML=win?`✅ ¡Correcto! Obtuviste <b>${sc}</b> puntos. ¡Excelente trabajo, explorador!`:`❌ La respuesta era <b>${correct}</b>. ¡Sigue explorando para aprender más!`;

}

const MOBS=[{v:'Investigación científica',ok:true},{v:'Terraformación inmediata',ok:false},{v:'Despliegue de satélite',ok:true}];
let mSel={dest:null,nave:null,obj:null};
function startMission(){
  mSel={dest:null,nave:null,obj:null};
  goTo('missionScreen');
  renderMission();
  openSteps([{p:'🦐',n:'SHRIMPY',sub:'Camarón astronauta',badge:'',
    t:'¡Tiempo de actuar! ¡Diseña tu propia misión espacial! Elige el destino, el tipo de nave y los objetivos. ¡Capi analizará si tu misión puede tener éxito basándose en ciencia real!',
    ch:[{l:'¡Comenzar diseño!',cb:closeDlg}]}]);
}
function renderMission(){
  const area=document.getElementById('missionArea');
  area.innerHTML=`
    <h2 class="sec-title" style="margin:0;">🛸 Diseña tu misión espacial</h2>
    <div class="mission-step">
      <div class="ms-title">🪐 PASO 1 — Elige el destino</div>
      <div class="ms-opts">${MDESTS.map(d=>`<button class="ms-opt ${mSel.dest===d.v?'sel':''}" onclick="selM('dest','${d.v}',this)">${d.v}</button>`).join('')}</div>
    </div>
    <div class="mission-step">
      <div class="ms-title">🚀 PASO 2 — Elige el tipo de nave</div>
      <div class="ms-opts">${MNAVES.map(n=>`<button class="ms-opt ${mSel.nave===n.v?'sel':''}" onclick="selM('nave','${n.v}',this)">${n.v}</button>`).join('')}</div>
    </div>
    <div class="mission-step">
      <div class="ms-title">🎯 PASO 3 — Elige el objetivo</div>
      <div class="ms-opts">${MOBS.map(o=>`<button class="ms-opt ${mSel.obj===o.v?'sel':''}" onclick="selM('obj','${o.v}',this)">${o.v}</button>`).join('')}</div>
    </div>
    <button class="mission-launch" id="mLaunch" onclick="launchMission()" ${(!mSel.dest||!mSel.nave||!mSel.obj)?'disabled':''}>🚀 ¡LANZAR MISIÓN!</button>
    <div class="mission-result" id="mResult"></div>`;
}
function selM(k,v,btn){
  mSel[k]=v;
  btn.closest('.ms-opts').querySelectorAll('.ms-opt').forEach(b=>b.classList.remove('sel'));
  btn.classList.add('sel');
  document.getElementById('mLaunch').disabled=!mSel.dest||!mSel.nave||!mSel.obj;
}
function launchMission(){
  const d=MDESTS.find(x=>x.v===mSel.dest);
  const n=MNAVES.find(x=>x.v===mSel.nave);
  const o=MOBS.find(x=>x.v===mSel.obj);
  const success=d.ok&&n.ok&&o.ok;
  const r=document.getElementById('mResult');
  r.className='mission-result show';
  r.innerHTML=success
    ?`<div class="mr-icon">🎉</div><div class="mr-title">¡Misión exitosa!</div><div class="mr-desc">Destino: ${d.v} — ${d.sci}<br><br>Nave: ${n.v} — ${n.sci}<br><br>Objetivo: ${o.v} — ¡Completado con éxito!</div>`
    :`<div class="mr-icon">💥</div><div class="mr-title">Misión fallida</div><div class="mr-desc">${!d.ok?'⚠️ Destino: '+d.sci+'<br><br>':''} ${!n.ok?'⚠️ Nave: '+n.sci+'<br><br>':''} ${!o.ok?'⚠️ Objetivo inviable con la tecnología actual.':''}</div>`;
  setTimeout(()=>{
    openSteps([{p:'🦫',n:'CAPI',sub:'Capibara astronauta',badge:'',
      t:success?`¡Excelente planificación, explorador! ${d.sci} Esta combinación de destino, nave y objetivo es científicamente viable según los datos actuales de la NASA.`:
        `La misión no tuvo éxito. ${!d.ok?d.sci:''} ${!n.ok?n.sci:''} ${!o.ok?'El objetivo no es técnicamente alcanzable con la tecnología actual.':''} ¡Sigue intentando!`,
      ch:[{l:'Rediseñar misión',cb:()=>{closeDlg();renderMission();}},{l:'Volver',cb:()=>{closeDlg();goTo('games');}}]}]);
  },1200);
}
